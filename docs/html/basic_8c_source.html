507 /* string variable */

521 else if(kw("LCASE$")) { fn_lcase(tmp); }
522 else if(kw("SPACE$")) { fn_space(tmp); }
523 else if(kw("STRING$")) { fn_string(tmp); }
524 else if(kw("INKEY$")) { fn_inkey(tmp); }
525 else {

527 int32_t v=parse_expr();
528 /* convert to string */
529 char nbuf[20]; int ni=0;

587 if(kw("VAL")) { return fn_val(); }

589 if(kw("ASC")) { return fn_asc(); }
590 if(kw("INSTR")) { return fn_instr(); }
591
592 /* Array variable A(n) */
593 if(*gp>='A'&&*gp<='Z'&&*(gp+1)=='('&&*(gp+2)!='$') {
594 int idx=*gp-'A'; gp+=2;
595 int32_t subscr=parse_expr();
596 skip_sp(); if(*gp==')') gp++;
597 if(!arr_vars[idx]||subscr<0||subscr>=arr_size[idx]) {
598 kprintf("?Subscript out of range\n"); return 0;
599 }
600 return arr_vars[idx][subscr];
601 }
602
603 /* Integer variable A–Z (or A% style) */
604 if(*gp>='A'&&*gp<='Z') {
605 /* Check it's not a string var or array */
606 if(*(gp+1)!='$'&&*(gp+1)!='(') {
607 int idx=*gp-'A'; gp++;
608 if(*gp=='%') gp++; /* optional type suffix */
609 /* Check constants table */
610 char name[2]={(char)('A'+idx),'\0'};
611 for(int i=0;i<const_count;i++) {
612 if(bstrcmp(consts[i].name,name)==0) return consts[i].val;
613 }
614 return int_vars[idx];
615 }
616 }
617
618 /* Hex literal &H */
619 if(*gp=='&'&&*(gp+1)=='H') {
620 gp+=2;
621 int32_t v=0;
622 while((*gp>='0'&&*gp<='9')||(*gp>='A'&&*gp<='F')||(*gp>='a'&&*gp<='f')) {
623 int d=(*gp>='a')?*gp-'a'+10:(*gp>='A')?*gp-'A'+10:*gp-'0';
624 v=v*16+d; gp++;
625 }
626 return v;
627 }
628
629 /* Octal literal &O */
630 if(*gp=='&'&&*(gp+1)=='O') {
631 gp+=2;
632 int32_t v=0;
633 while(*gp>='0'&&*gp<='7'){v=v*8+(*gp++-'0');}
634 return v;
635 }
636
637 /* Decimal literal */
638 int32_t v=0; bool got=false;
639 while(*gp>='0'&&*gp<='9'){v=v*10+(*gp++-'0');got=true;}
640 if(*gp=='.') { gp++; while(*gp>='0'&&*gp<='9') gp++; } /* skip frac */
641 if(!got) return 0;
642 return v;
643}

798/* Split a line into colon-separated statements.
799 Returns pointer to stmt #n within line text (skip line number first). */
800static const char *get_stmt(const char *text, int n)
801{

809 else p++;
810 }
811 if(!*p) return NULL;
812 p++; s++;
813 while(*p==' ') p++;
814 }
815 return p;
816}

1158static void exec_if_single(int32_t cond)
1159{

1161 const char *else_pos=NULL;
1162 const char *scan=gp;
1163 int depth=0;
1164 while(*scan) {
1165 if(*scan=='"'){scan++;while(*scan&&*scan!='"')scan++;if(*scan)scan++;continue;}
1166 if(bstrncmp(scan,"IF",2)==0&&(scan[2]==' '||scan[2]=='(')) depth++;
1167 if(bstrncmp(scan,"END IF",6)==0) depth--;
1168 if(depth==0&&bstrncmp(scan,"ELSE",4)==0&&
1169 (scan[4]==' '||scan[4]=='\0'||scan[4]=='\r')) { else_pos=scan; break; }
1170 scan++;
1171 }
1172
1173 if(cond) {
1174 /* execute THEN part (gp is already past THEN keyword) */
1175 /* truncate at ELSE if found: temporarily NUL-terminate... */
1176 /* Actually we just stop at ELSE by setting a boundary: use skip_to_colon logic */
1177 /* For simplicity: exec statements until we hit ELSE or EOL */
1178 /* We re-enter exec_stmt logic inline */
1179 /* QB: THEN can be followed by a line number (implicit GOTO) */
1180 skip_sp();
1181 if(*gp>='0'&&*gp<='9') {
1182 int ln=(int)parse_expr();

1184 if(idx<0){kprintf("?Undefined line %d\n",ln);stopped=true;}
1185 else jump_to(idx);

1187 }
1188 /* otherwise the rest of gp until ELSE is the then-body;
1189 we'll just leave gp where it is and let the main dispatch handle it
1190 by setting a "else_skip" flag — but that's complex.
1191 Simpler: parse ONE statement, then handle ELSE ourselves. */
1192 /* Leave gp at then-body; caller (exec_line) will dispatch next stmt.
1193 We mark that ELSE should be skipped. */
1194 /* For now: just skip ELSE branch */
1195 (void)else_pos;
1196 /* gp stays at THEN body — let exec_line dispatch it */
1197 } else {
1198 if(else_pos) {
1199 gp=else_pos+4; /* skip "ELSE" */
1200 skip_sp();
1201 } else {
1202 /* skip rest of line */
1203 while(*gp) gp++;
1204 }
1205 }

1212 if_stack[if_top].active=(cond!=0);
1213 if_stack[if_top].done=(cond!=0);
1214 if_stack[if_top].else_seen=0;

1216 if(!cond) {
1217 /* skip to ELSE / ELSEIF / END IF */

1316static void exec_case(void)
1317{
1318 if(sel_top<=0){kprintf("?CASE without SELECT\n");stopped=true;return;}
1319 sel_frame_t *f=&sel_stack[sel_top-1];
1320 if(f->done) {
1321 /* skip to END SELECT */
1322 int depth=1;
1323 while(cur_idx<prog_count&&depth>0) {
1324 if(!advance_stmt()){cur_idx++;cur_stmt=0;if(cur_idx>=prog_count)break;}
1325 const char *p=get_stmt(prog[cur_idx].text,cur_stmt);
1326 if(!p)continue;
1327 while(*p==' ')p++;
1328 if(bstrncmp(p,"SELECT",6)==0) depth++;
1329 else if(bstrncmp(p,"END SELECT",10)==0) { depth--; if(!depth) break; }
1330 }
1331 sel_top--;
1332 return;
1333 }
1334 skip_sp();
1335 /* CASE ELSE */
1336 if(kw("ELSE")) { f->matched=1; f->done=1; return; }
1337 /* CASE IS rel expr */
1338 bool match=false;
1339 if(kw("IS")) {
1340 skip_sp();
1341 int32_t v=f->val, rhs;
1342 if(kw_any("<>")){ rhs=parse_expr(); match=(v!=rhs); }
1343 else if(kw_any("<=")){ rhs=parse_expr(); match=(v<=rhs); }
1344 else if(kw_any(">=")){ rhs=parse_expr(); match=(v>=rhs); }
1345 else if(*gp=='<'){ gp++; rhs=parse_expr(); match=(v<rhs); }
1346 else if(*gp=='>'){ gp++; rhs=parse_expr(); match=(v>rhs); }
1347 else if(*gp=='='){ gp++; rhs=parse_expr(); match=(v==rhs); }
1348 } else {
1349 /* CASE val [, val ...] or CASE a TO b */
1350 do {
1351 skip_sp();
1352 int32_t lo=parse_expr();
1353 skip_sp();
1354 if(kw("TO")) {
1355 int32_t hi=parse_expr();

1357 } else {
1358 if(f->val==lo) match=true;
1359 }
1360 skip_sp();
1361 } while(*gp==',' && gp++);
1362 }
1363 if(match) { f->matched=1; f->done=1; }
1364 else {
1365 /* skip to next CASE or END SELECT */
1366 while(cur_idx<prog_count) {
1367 if(!advance_stmt()){cur_idx++;cur_stmt=0;if(cur_idx>=prog_count)break;}
1368 const char *p=get_stmt(prog[cur_idx].text,cur_stmt);
1369 if(!p)continue;
1370 while(*p==' ')p++;
1371 if(bstrncmp(p,"CASE",4)==0) break;
1372 if(bstrncmp(p,"END SELECT",10)==0){sel_top--;return;}
1373 }
1374 }
