179static int str_tokenize(char *str, char **tokens, int max_tokens)
180{
181 int count = 0;
182 char *p = str;
183
184 while (*p && count < max_tokens) {
185 while (*p == ' ' || *p == '\t')
186 p++;
187
188 if (*p == '\0')
189 break;
190
191 tokens[count++] = p;
192
193 while (*p && *p != ' ' && *p != '\t')
194 p++;
195
196 if (*p)
197 *p++ = '\0';
198 }
199
200 return count;
201}

406 cmd_gui();
407 }

409 uart_puts("\r\nBad command or file name: ");

412 }
