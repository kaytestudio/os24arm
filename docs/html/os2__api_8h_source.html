1/* ============================================================================
2 * os2api/os2_api.h — OS/2 DOS-call API
3 *
4 * A DOS-call-shaped bridge onto the kernel's real VFS layer (vfs_open,
5 * vfs_read, vfs_opendir, ...). This is the layer dos/ command
6 * implementations go through to touch the filesystem, rather than calling
7 * vfs_* directly - matching how real OS/2 programs call DOSCALLS.DLL
8 * instead of talking to the filesystem driver themselves.
9 * ========================================================================== */
