21int blkdev_register(blkdev_t *dev)
22{
23 uart_puts("DEBUG: Entering blkdev_register\r\n");
24 if (blkdev_count >= MAX_BLKDEVS) return -1;
25 blkdevs[blkdev_count++] = dev;
26 kprintf("[BLKDEV] Registered: %s (%u sectors)\n",

28 uart_puts("DEBUG: blkdev_register complete\r\n");

63static int ramdisk_read(blkdev_t *dev, int64_t lba, uint32_t count, void *buf)
64{
65 ramdisk_priv_t *p = (ramdisk_priv_t *)dev->driver_data;
66 if (!p || !p->base) return -1;
67 if (lba + count > p->num_sectors) return -1;
68
69 kmemcpy(buf, p->base + lba * SECTOR_SIZE, count * SECTOR_SIZE);
70 return 0;
71}

83void ramdisk_create(const char *name, void *base, size_t size)
84{
85 ramdisk_priv.base = (uint8_t *)base;
86 ramdisk_priv.num_sectors = size / SECTOR_SIZE;
87
88 kstrcpy(ramdisk_dev.name, name);
89 ramdisk_dev.num_sectors = ramdisk_priv.num_sectors;
90 ramdisk_dev.sector_size = SECTOR_SIZE;
91 ramdisk_dev.driver_data = &ramdisk_priv;
92 ramdisk_dev.read = ramdisk_read;
93 ramdisk_dev.write = ramdisk_write;
94 ramdisk_dev.flags = 0;
95 ramdisk_dev.parent = NULL;
96 ramdisk_dev.start_sector = 0;
97
98 blkdev_register(&ramdisk_dev);
99 kprintf("[RAMDISK] Created: %s (%u KB)\n", name, (unsigned)(size / 1024));
100}

163/* ── Block Device Subsystem Init ───────────────────────────────────────── */

165/* No VirtIO block driver exists yet (see virtio_blk_read/write stubs
166 * above), so there's no real disk to mount a filesystem against. Register
167 * a block device backed by a fixed physical RAM address instead: a real
168 * disk image can be tested by injecting it there before boot, e.g.
169 * qemu-system-aarch64 ... -device loader,file=test.img,addr=0x44000000
170 * With nothing loaded there, RAM reads as zero and filesystem mounts
171 * correctly fail (bad superblock magic) rather than misbehaving. */
172#define PRELOADED_DISK_ADDR 0x44000000UL
173#define PRELOADED_DISK_SIZE (24u * 1024u * 1024u)
174
175void blk_init(void)
176{
177 uart_puts("DEBUG: Entering blk_init\r\n");
178
179 // Arrays are already zeroed by boot.S BSS clearing
180 blkdev_count = 0;
181
182 ramdisk_create("disk0", (void *)PRELOADED_DISK_ADDR, PRELOADED_DISK_SIZE);
183
184 /* If a GPT-partitioned image (BOOT + C:, or anything else) was loaded
185 * into disk0, register its partitions as child devices. No-op if disk0
186 * doesn't contain a valid GPT (e.g. a flat, unpartitioned test image),
187 * leaving disk0 itself directly usable as before. */
188 gpt_probe(blkdev_get_by_name("disk0"));
189
190 uart_puts("DEBUG: blk_init complete\r\n");
191}
